  function leerHash(){
    var h = location.hash || '';
    if(h.charAt(0) === '#') h = h.slice(1);
    if(!h) return null;
    var p = {};
    h.split('&').forEach(function(par){
      var kv = par.split('=');
      if(kv[0]) p[decodeURIComponent(kv[0])] = decodeURIComponent(kv[1] || '');
    });
    return p.access_token ? p : null;
  }

        if(salir) salir.addEventListener('click', function(e){ e.preventDefault(); window.cerrarSesionComprender(); });

        // Corte L: solo existe cuando perfilBajaProgramada esta seteado (ver pintar()).
        var cancelarBaja = capa.querySelector('#ccCancelarBaja');
        if(cancelarBaja){
          cancelarBaja.addEventListener('click', function(){
            var avisoEl2 = capa.querySelector('#ccAviso');
            cancelarBaja.disabled = true; cancelarBaja.textContent = 'Cancelando…';
            tokenVigente().then(function(t){
              if(!t){
                if(avisoEl2) avisoEl2.textContent = 'Tu sesión venció. Cerrá esto y volvé a entrar.';
                cancelarBaja.disabled = false; cancelarBaja.textContent = 'Cancelar eliminación';
                return;
              }
              // Corte M (05/08): /api/cancelar-baja se unifico en /api/cuenta (accion en el
              // body) por el limite de 12 Funciones Serverless de Vercel -- ver la nota en
              // la cabecera de api/cuenta.js.
              _fetch('/api/cuenta', {
                method:'POST',
                headers:{ 'content-type':'application/json', Authorization:'Bearer ' + t },
                body: JSON.stringify({ accion:'cancelar_baja' })
              })
                .then(function(r){
                  if(r.ok){
                    perfilBajaProgramada = null;
                    mensajeCuenta = 'Listo, tu cuenta ya no está programada para eliminarse.';
                    pintar();
                    return;
                  }
                  return r.json().catch(function(){ return {}; }).then(function(d){
                    cancelarBaja.disabled = false; cancelarBaja.textContent = 'Cancelar eliminación';
                    if(avisoEl2) avisoEl2.textContent = (d && d.error && d.error.message) || 'No se pudo cancelar.';
                  });
                }).catch(function(){
                  cancelarBaja.disabled = false; cancelarBaja.textContent = 'Cancelar eliminación';
                  if(avisoEl2) avisoEl2.textContent = 'No se pudo conectar. Revisá tu conexión.';
                });
            });
          });
        }
        return;
      }

      var avisoEl = capa.querySelector('#ccAviso');
      var accion = capa.querySelector('#ccAccion');
      capa.querySelector('#ccVolver').addEventListener('click', function(e){ e.preventDefault(); vistaC='inicio'; pintar(); });

      function ocupado(si, texto){ accion.disabled = si; if(texto) accion.textContent = texto; }

      if(vistaC === 'clave'){
        var actual = capa.querySelector('#ccClaveActual'), nueva = capa.querySelector('#ccClaveNueva');
        accion.addEventListener('click', function(){
          var k1 = actual.value||'', k2 = nueva.value||'';
          if(!k1||!k2){ avisoEl.textContent='Faltan datos.'; return; }
          if(k2.length<6){ avisoEl.textContent='La contraseña nueva necesita al menos 6 caracteres.'; return; }
          ocupado(true,'Guardando…'); avisoEl.textContent='';
          ingresar(cuentaCorreo, k1).then(function(r){
            if(!r.ok){ ocupado(false,'Guardar'); avisoEl.textContent = 'Contraseña actual incorrecta.'; return; }
            var s = leer();
            cambiarClave(s.token, k2).then(function(r2){
              if(r2.ok){ mensajeCuenta = 'Contraseña actualizada.'; vistaC='inicio'; pintar(); }
              else{ ocupado(false,'Guardar'); avisoEl.textContent = r2.mensaje; }
            });
          });
        });
      } else if(vistaC === 'eliminar'){
        var confirmar = capa.querySelector('#ccConfirmar');
        accion.addEventListener('click', function(){
          if((confirmar.value||'').trim().toUpperCase() !== 'ELIMINAR'){
            avisoEl.textContent = 'Escribí ELIMINAR, tal cual, para confirmar.'; return;
          }
          ocupado(true,'Eliminando…'); avisoEl.textContent='';
          tokenVigente().then(function(t){
            if(!t){ avisoEl.textContent='Tu sesión venció. Cerrá esto y volvé a entrar.'; ocupado(false,'Eliminar mi cuenta'); return; }
            // Corte M (05/08): /api/eliminar-cuenta se unifico en /api/cuenta (accion en el
            // body) por el limite de 12 Funciones Serverless de Vercel -- ver la nota en la
            // cabecera de api/cuenta.js. 'eliminar' es default si no se manda body, pero se
            // manda explicito igual, por claridad.
            _fetch('/api/cuenta', {
              method:'POST',
              headers:{ 'content-type':'application/json', Authorization:'Bearer ' + t },
              body: JSON.stringify({ accion:'eliminar' })
            }).then(function(r){
              if(r.ok){
                // Corte L: ya no borra al instante -- queda programada. Se sigue con la
                // sesion abierta (el usuario puede seguir usando la cuenta hasta la fecha,
                // o arrepentirse desde este mismo panel).
                return r.json().catch(function(){ return {}; }).then(function(d){
                  perfilBajaProgramada = (d && d.baja_programada) || null;
                  mensajeCuenta = 'Listo, tu cuenta quedó programada para eliminarse en 7 días. Podés cancelarlo cuando quieras desde este panel.';
                  vistaC = 'inicio';

                });

                ocupado(false,'Eliminar mi cuenta');
                avisoEl.textContent = (d && d.error && d.error.message) || 'No se pudo eliminar la cuenta.';
              });
            }).catch(function(){
              ocupado(false,'Eliminar mi cuenta');
              avisoEl.textContent = 'No se pudo conectar. Revisá tu conexión.';
            });
          });
        });
      }
    }

    pintar();   // primer pintado inmediato ("Cargando…"), se repinta solo cuando llega el correo
    tokenVigente().then(function(t){
      if(!t) return;
      obtenerUsuario(t).then(function(u){
        cuentaCorreo = (u && u.email) || '';
        if(vistaC === 'inicio') pintar();
      });
    });

  /* ---------- bandeja (Corte B del sistema comunicacional, 03/08) ----------
     Contador + panel de avisos internos. Vive aca por la misma razon que "Cuenta" y los
     planes: es infraestructura compartida por los tres productos, no se repite en cada
     HTML. Lee y marca via api/comm.js (recurso=inbox) -- no hay todavia ningun productor real
     conectado (eso es un corte posterior), asi que puede aparecer vacia; el contador y
     el panel ya quedan listos para cuando existan avisos de verdad. */
  function bandejaFetch(ruta, opciones){
    opciones = opciones || {};
    return tokenVigente().then(function(t){
      if(!t) return Promise.reject(new Error('sin sesión'));
      var headers = opciones.headers || {};
      headers['Authorization'] = 'Bearer ' + t;
      headers['content-type'] = 'application/json';
      opciones.headers = headers;
      return _fetch(ruta, opciones);
    });
  }

      });
    });
  };

  /* ---------- arranque ----------
     Antes de mirar si hay sesion guardada, se revisa si el navegador volvio con un
     link de correo (hash con access_token). Si es de recuperacion, se muestra la
     pantalla de nueva contrasena en vez del portal normal. Cualquier otro hash con
     token (confirmacion de alta, magic link, invitacion) deja una sesion valida:
     se guarda y arranca directo, sin pasar por el portal.

     Corte AD (09/08): la deteccion de recuperacion (query ?type=recovery o hash
     #access_token...&type=recovery) se saca de arrancar() y corre ACA MISMO, sincronica
     e inmediata -- no puede esperar a DOMContentLoaded. El segundo <script> de cada
     pagina real (index/urbanismo/contextos) dispara su propio fetch de arranque
     (/api/organismos) apenas se parsea; sin sesion guardada, tokenVigente() resuelve casi
     instantaneo (sin red, ver mas arriba) y ese fetch le ganaba la carrera al portal de
     recuperacion -- se terminaba mostrando "Tu sesion vencio" tapando la pantalla de
     "Elegi una contraseña nueva" (confirmado en vivo con diagnostico en sessionStorage:
     el 401 llegaba unos milisegundos antes que arrancar(), que estaba esperando
     DOMContentLoaded). document.body ya existe en este punto (este script esta adentro
     de <body>), asi que portal() puede llamarse ya mismo.

     Corte AE (09/08): mismo mecanismo (link a esta pagina + query token_hash, canje recien
     con un clic real) para la confirmacion de alta, y por la misma razon (la carrera contra
     el fetch de arranque) se agrega ACA, no adentro de arrancar(). */
  var _arranqueInicial = (function(){
    var th = leerQueryRecovery();
    if(th){ limpiarQuery(); return { modo: 'nueva_clave', extra: { token_hash: th } }; }
    var ts = leerQuerySignup();
    if(ts){ limpiarQuery(); return { modo: 'confirmar_cuenta', extra: { token_hash: ts } }; }
    var h = leerHash();
    if(h && h.type === 'recovery'){ limpiarHash(); return { modo: 'nueva_clave', extra: { token: h.access_token, refresco: h.refresh_token || '' } }; }
    return null;
  })();
  if(_arranqueInicial) portal('', _arranqueInicial.modo, _arranqueInicial.extra);

  function arrancar(){
    if(_arranqueInicial) return;
    var h = leerHash();
    if(h && h.access_token){
      guardar({
        token: h.access_token,
        refresco: h.refresh_token || '',
        vence: Date.now() + ((Number(h.expires_in) || 3600) * 1000),
        correo: ''
      });
      limpiarHash();
      sincronizarSaldo();
      return;
    }
    tokenVigente().then(function(t){
      if(!t) portal('');
      else sincronizarSaldo();
    });
  }
  if(document.readyState === 'loading') document.addEventListener('DOMContentLoaded', arrancar);
  else arrancar();

  /* ---------- envoltorio de fetch ---------- */
  var _fetch = window.fetch;
  window.fetch = function(recurso, opciones){
    var url = '';
    try{ url = String((recurso && recurso.url) ? recurso.url : recurso || ''); }catch(e){}
    var esApiComprender = url.indexOf('/api/anthropic') > -1 || url.indexOf('/api/organismos') > -1;
    if(!esApiComprender) return _fetch.apply(this, arguments);

    var args = arguments;
    var yo = this;
    return tokenVigente().then(function(t){
      if(!t){ portal('Tu sesión venció. Ingresá de nuevo.');
              return Promise.reject(new Error('sin sesión')); }

      opciones = opciones || {};
      var h = opciones.headers, plano = {};
      try{
        if(h && typeof h.forEach === 'function' && !Array.isArray(h)){ h.forEach(function(v,k){ plano[k]=v; }); }
        else if(h){ for(var k in h){ if(Object.prototype.hasOwnProperty.call(h,k)) plano[k]=h[k]; } }
      }catch(e){}
      plano['Authorization'] = 'Bearer ' + t;
      try{ plano['x-comprender-modulo'] = (typeof MODULE_ID === 'string' && MODULE_ID) ? MODULE_ID : 'core'; }
      catch(e){ plano['x-comprender-modulo'] = 'core'; }
      opciones.headers = plano;

      return _fetch.call(yo, recurso, opciones).then(function(resp){
        try{
          var g = resp && resp.headers && resp.headers.get ? resp.headers : null;
          if(g) anotarSaldo(g.get('x-comprender-plan'), g.get('x-comprender-saldo'));
        }catch(e){}
        // 401 del proxy = la sesion no sirve. Se limpia y se vuelve a pedir.
        if(resp && resp.status === 401){ borrar(); portal('Tu sesión venció. Ingresá de nuevo.'); }

        // 402 sin saldo · 403 plan insuficiente: el contador es el disparador,
        // la pantalla de planes es el destino. Se abre sola donde hizo falta.
        if(resp && (resp.status === 402 || resp.status === 403)){
          try{
            resp.clone().json().then(function(d){
              var m = (d && d.error && d.error.codigo) || '';
              if(m === 'sin_saldo' || m === 'requiere_plan') window.mostrarPlanes(m);
            }).catch(function(){});
          }catch(e){}
        }
        return resp;
      });
    });
  };
