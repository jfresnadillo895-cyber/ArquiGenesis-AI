  function desdeRespuesta(d){
    if(!d || !d.access_token) return null;
    return {
      token:   d.access_token,
      refresco: d.refresh_token || '',
      vence:   Date.now() + ((d.expires_in || 3600) * 1000),

  function auth(ruta, cuerpo){
    return fetch(SB_URL + '/auth/v1/' + ruta, {
      method: 'POST',
      headers: { apikey: SB_KEY, 'content-type': 'application/json' },
      body: JSON.stringify(cuerpo)
    }).then(function(r){
      return r.json().then(function(d){ return { ok: r.ok, estado: r.status, datos: d }; });
    });

  function ingresar(correo, clave){
    return auth('token?grant_type=password', { email: correo, password: clave })
      .then(function(r){
        if(!r.ok) return { ok:false, mensaje: mensajeAuth(r) };
        var s = desdeRespuesta(r.datos);
        if(!s) return { ok:false, mensaje:'Respuesta inesperada del servidor.' };
        guardar(s);
        return { ok:true };
      })
      .catch(function(){ return { ok:false, mensaje:'No se pudo contactar al servidor. Revisá tu conexión.' }; });

  function mensajeAuth(r){
    var d = r.datos || {};

    if(r.estado === 400 || r.estado === 401) return 'Correo o contraseña incorrectos.';
    if(t.indexOf('confirm') > -1) return 'La cuenta todavía no está confirmada.';
    if(r.estado >= 500) return 'El servicio no está disponible en este momento. Probá en unos minutos.';
    return 'No se pudo iniciar sesión.';
  }

  function leerHash(){
    var h = location.hash || '';
    if(h.charAt(0) === '#') h = h.slice(1);
    if(!h) return null;
    var p = {};
    h.split('&').forEach(function(par){
      var kv = par.split('=');
      if(kv[0]) p[decodeURIComponent(kv[0])] = decodeURIComponent(kv[1] || '');
    });
    return p.access_token ? p : null;
  }
  function limpiarHash(){
    try{ history.replaceState(null, '', location.pathname + location.search); }catch(e){}

      if(vistaC === 'inicio'){
        capa.querySelector('#ccIrClave').addEventListener('click', function(){ vistaC='clave'; pintar(); });
        capa.querySelector('#ccIrEliminar').addEventListener('click', function(){ vistaC='eliminar'; pintar(); });

          if(!k1||!k2){ avisoEl.textContent='Faltan datos.'; return; }
          if(k2.length<6){ avisoEl.textContent='La contraseña nueva necesita al menos 6 caracteres.'; return; }
          ocupado(true,'Guardando…'); avisoEl.textContent='';
          ingresar(cuentaCorreo, k1).then(function(r){
            if(!r.ok){ ocupado(false,'Guardar'); avisoEl.textContent = 'Contraseña actual incorrecta.'; return; }
            var s = leer();
            cambiarClave(s.token, k2).then(function(r2){
              if(r2.ok){ mensajeCuenta = 'Contraseña actualizada.'; vistaC='inicio'; pintar(); }
              else{ ocupado(false,'Guardar'); avisoEl.textContent = r2.mensaje; }
            });
          });
        });

          ocupado(true,'Eliminando…'); avisoEl.textContent='';

              });

              avisoEl.textContent = 'No se pudo conectar. Revisá tu conexión.';
            });
          });
        });
      }

  /* ---------- arranque ----------
     Antes de mirar si hay sesion guardada, se revisa si el navegador volvio con un
     link de correo (hash con access_token). Si es de recuperacion, se muestra la
     pantalla de nueva contrasena en vez del portal normal. Cualquier otro hash con
     token (confirmacion de alta, magic link, invitacion) deja una sesion valida:
     se guarda y arranca directo, sin pasar por el portal. */
  function arrancar(){
    var h = leerHash();
    if(h && h.type === 'recovery'){
      limpiarHash();
      portal('', 'nueva_clave', { token: h.access_token, refresco: h.refresh_token || '' });
      return;
    }
    if(h && h.access_token){
      guardar({
        token: h.access_token,
        refresco: h.refresh_token || '',
        vence: Date.now() + ((Number(h.expires_in) || 3600) * 1000),
        correo: ''
      });
      limpiarHash();
      sincronizarSaldo();
      return;
    }
    tokenVigente().then(function(t){
      if(!t) portal('');
      else sincronizarSaldo();
    });
  }
  if(document.readyState === 'loading') document.addEventListener('DOMContentLoaded', arrancar);

    var esApiComprender = url.indexOf('/api/anthropic') > -1 || url.indexOf('/api/organismos') > -1;
    if(!esApiComprender) return _fetch.apply(this, arguments);

    var args = arguments;
    var yo = this;
    return tokenVigente().then(function(t){

              return Promise.reject(new Error('sin sesión')); }

      opciones = opciones || {};
      var h = opciones.headers, plano = {};
      try{
        if(h && typeof h.forEach === 'function' && !Array.isArray(h)){ h.forEach(function(v,k){ plano[k]=v; }); }
        else if(h){ for(var k in h){ if(Object.prototype.hasOwnProperty.call(h,k)) plano[k]=h[k]; } }

        }catch(e){}
        // 401 del proxy = la sesion no sirve. Se limpia y se vuelve a pedir.

              if(m === 'sin_saldo' || m === 'requiere_plan') window.mostrarPlanes(m);
            }).catch(function(){});
          }catch(e){}
        }
        return resp;
      });
    });
